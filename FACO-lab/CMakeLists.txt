cmake_minimum_required(VERSION 3.16)
project(faco_lab VERSION 1.0 LANGUAGES CXX)
option(FACO_OPENMP "Enable OpenMP" ON)
option(FACO_SANITIZERS "Address/undefined sanitizers (GCC/Clang on Linux)" OFF)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(faco_core src/problem.cpp src/neighbors.cpp src/tour.cpp src/pheromone.cpp src/local_search.cpp src/faco.cpp)
target_include_directories(faco_core PUBLIC include)
if(MSVC)
  target_compile_options(faco_core PUBLIC /W4)
else()
  target_compile_options(faco_core PUBLIC -Wall -Wextra -Wpedantic -Wshadow)
endif()
if(FACO_OPENMP)
  find_package(OpenMP REQUIRED)
  target_link_libraries(faco_core PUBLIC OpenMP::OpenMP_CXX)
endif()
if(FACO_SANITIZERS)
  if(WIN32 OR MSVC)
    message(FATAL_ERROR "FACO_SANITIZERS requires GCC/Clang on Linux")
  endif()
  target_compile_options(faco_core PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(faco_core PUBLIC -fsanitize=address,undefined)
endif()
add_executable(faco src/main.cpp src/options.cpp src/output.cpp)
target_link_libraries(faco PRIVATE faco_core)
enable_testing()
add_executable(faco_tests tests/tests.cpp)
target_link_libraries(faco_tests PRIVATE faco_core)
target_compile_definitions(faco_tests PRIVATE FACO_DATA_DIR="${CMAKE_CURRENT_SOURCE_DIR}/data")
add_test(NAME faco_mechanisms COMMAND faco_tests)
add_test(NAME cli_help COMMAND faco --help)
add_test(NAME cli_reject_unknown COMMAND faco --unknown 1)
set_tests_properties(cli_reject_unknown PROPERTIES WILL_FAIL TRUE)
